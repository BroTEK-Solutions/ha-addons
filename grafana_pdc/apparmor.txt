#include <tunables/global>

# Reads stay broad so diagnostics cannot silently lose data. Writes are limited
# to s6 runtime files and the App's own persistent SSH material. The ingress UI
# and MQTT reporter inherit this outer profile; the PDC agent transitions below.
#
# Recovery: a missing permission can prevent a service from starting, make the
# status page unavailable, or leave MQTT entities unavailable. Temporarily set
# `apparmor: false` in grafana_pdc/config.yaml while diagnosing the denied path
# in the HAOS host's `dmesg`; restore confinement after correcting the policy.
profile grafana_pdc flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  # A bare file rule grants writes everywhere; later rules cannot restrict it.
  /{,**} rm,
  signal (send) set=(kill,term,int,hup,cont),
  # Container runtime termination; do not grant signals from arbitrary peers.
  signal (receive) set=(kill,term,int,hup,cont) peer=unconfined,
  signal (receive) set=(kill,term,int,hup,cont) peer=runc,
  capability chown,
  capability kill,
  capability setgid,
  capability setuid,

  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  # curl runs under this profile for the IPv4 loopback health probe. The status
  # page and the MQTT reporter also run here: the page dials allowlisted
  # destinations to test reachability, which may be IPv6, and the reporter
  # connects to the Home Assistant broker.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /dev/null rw,

  # Bashio and the add-on's transient setup files.
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Persistent state is limited to SSH material managed by the service.
  /data/options.json r,
  /data/ssh/ rwk,
  /data/ssh/** rwk,

  # The service gets its own constrained profile.
  /usr/bin/pdc cx -> pdc,

  profile pdc flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    signal (receive) peer=*_grafana_pdc,
    network,

    # Go maps executable segments from its own binary during startup.
    /usr/bin/pdc mr,
    /usr/bin/ssh ix,
    # Prometheus process metrics and OpenSSH close-from discovery.
    /proc/[0-9]*/{stat,limits,cgroup} r,
    /proc/[0-9]*/mountinfo r,
    /proc/[0-9]*/net/netstat r,
    /proc/[0-9]*/fd/ r,
    /proc/sys/net/core/somaxconn r,
    # Go runtime CPU quota discovery on cgroup v2.
    /sys/fs/cgroup/cpu.max r,
    /etc/ssh/** r,
    /etc/ssl/certs/** r,
    /etc/ssl/openssl.cnf r,
    /etc/passwd r,
    /etc/group r,
    /data/ssh/ rwk,
    /data/ssh/** rwk,
    /dev/null rw,
    /dev/urandom r,
  }
}
